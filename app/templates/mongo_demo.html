MongoDB Queries

Query 1: Events by Tag

db.event_content.find({ "tags": tag }, { "_id": 0, "eventId": 1, "title": 1, "tags": 1 }) 

        


        
        

            
Query 2: Events by Speaker

            

                db.event_content.find({ "speakers.name": name }, { "_id": 0 })
            

            

                
                
            

            

        


        
        

            
Query 3: Reviews Above Rating

            

                db.event_content.find({ "reviews.rating": { "$gt": min_rating } }, { "_id": 0, "eventId": 1, "title": 1, "reviews": 1 })
            

            

                
                
            

            

        


        
        

            
Query 4: Speaker by Org & Topic

            

                db.event_content.find({ "speakers": { "$elemMatch": { "organization": organization, "topics": topic } } }, { "_id": 0 })
            

            

                
                
                
            

            

        


        
        

            
Query 5: Concerts by Genre

            

                db.event_content.find({ "eventType": "Concert", "genres": genre }, { "_id": 0, "eventId": 1, "title": 1, "genres": 1 })
            

            

                
                
            

            

        


        
        

            
Query 6: Age Restricted OR Tagged

            

                db.event_content.find({ "$or": [{ "ageRestriction": { "$gte": min_age } }, { "tags": tag }] }, { "_id": 0, "eventId": 1, "title": 1, "ageRestriction": 1, "tags": 1 })
            

            

                
                
                
            

            

        


        
        

            
Query 7: Sessions in Room

            

                db.event_content.find({ "schedule.room": room }, { "_id": 0, "eventId": 1, "title": 1, "schedule": { "$elemMatch": { "room": room } } })
            

            

                
                
            

            

        


        
        

            
Query 8: Add Review ($push)

            

                db.event_content.updateOne({ "eventId": eventId }, { "$push": { "reviews": review } })
            

            

                
                
                
                
            

            

        


        
        

            
Query 9: Add Tag ($addToSet)

            

                db.event_content.updateOne({ "eventId": eventId }, { "$addToSet": { "tags": tag } })
            

            

                
                
                
            

            

        


        
        

            
Query 10: Delete Low Reviews ($pull with $lte)

            

                db.event_content.updateOne({ "eventId": eventId }, { "$pull": { "reviews": { "rating": { "$lte": max_rating } } } })
